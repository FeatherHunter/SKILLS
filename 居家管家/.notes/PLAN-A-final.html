def build_help_payload() -> dict:
    ...
    CATEGORY_ORDER = [
        "找东西","存东西","改东西","盘点","出门","回家",
        "看统计","账号","检查","标签","帮助"
    ]
    # 按 category 顺序 group,每个 category 内按 wake_word 分二级 group
    groups = []
    for cat in CATEGORY_ORDER:
        in_cat = [s for s in scenarios if s.get("category") == cat]
        if not in_cat:
            continue
        # 二级: 该 category 下的 wake_word 列表
        wake_words = []
        seen = set()
        for s in in_cat:
            if s["wake_word"] not in seen:
                seen.add(s["wake_word"])
                wake_words.append(s["wake_word"])
        groups.append({
            "category": cat,
            "wake_words": wake_words,
            "scenarios": in_cat,
        })
    return { ... "groups": groups ... }

def test_all_scenarios_have_category():
    valid = {"找东西","存东西","改东西","盘点","出门","回家",
             "看统计","账号","检查","标签","帮助"}
    for s in scenarios:
        assert s.get("category") in valid, f"{s['scenario_id']} 缺 category"
    # 7 字段契约(总纲 07 §2.2)
    for s in scenarios:
        for f in ["wake_word","scenario_id","scenario_title",
                  "category","dimensions","prompt","status","result"]:
            assert f in s, f"{s.get('scenario_id','?')} 缺 {f}"
