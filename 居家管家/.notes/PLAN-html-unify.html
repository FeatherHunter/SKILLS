1. scripts/build_manual.py — 同步脚本

"""把 HELP HTML 复制成 居家管家.html(总纲 04 §4.2 镜像源保证一致)

用法:
  python3 scripts/build_manual.py
  # 等价于:
  #   python3 scripts/home_manager.py help --output /tmp/help.html
  #   cp /tmp/help.html 居家管家.html
"""
import shutil, subprocess, sys
HELP_TMP = "/tmp/居家管家_help_generated.html"
r = subprocess.run([sys.executable, "scripts/home_manager.py",
                   "help", "--output", HELP_TMP],
                  capture_output=True, text=True)
if r.returncode != 0:
    sys.exit(r.returncode)
shutil.copy2(HELP_TMP, "居家管家.html")
print(f"✓ 居家管家.html 与 HELP HTML 同步完成")

2. tests/test_manual_sync.py — 同步断言

def test_manual_html_matches_help():
    """居家管家.html 必须与 HELP HTML 字节一致(scripts/build_manual.py 同步)"""
    import subprocess, hashlib, pathlib
    skill = pathlib.Path(__file__).parent.parent
    help_html = skill / "output" / "help_generated.html"
    manual = skill / "居家管家.html"
    # 生成 HELP HTML
    subprocess.run(["python3", str(skill/"scripts"/"home_manager.py"),
                    "help", "--output", str(help_html)], check=True)
    h1 = hashlib.sha256(help_html.read_bytes()).hexdigest()
    h2 = hashlib.sha256(manual.read_bytes()).hexdigest()
    # 容忍首行 banner 差异(可选:跳过 banner diff)
    assert h1 == h2, f"居家管家.html 与 HELP HTML 不一致:\\n{sha hint}"
